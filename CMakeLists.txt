cmake_minimum_required(VERSION 3.20)
project(c_mcp LANGUAGES C)

# The core and serial API's default stdio implementation use standard C.
option(CMCP_BUILD_SERIAL "Build the portable serial input loop" ON)
option(CMCP_BUILD_TESTS "Build portable dispatcher tests" OFF)
option(CMCP_ENABLE_VFS "Enable read-only MCP resources" ON)
add_library(CMCP cJSON.c mcp.c serial_transport.c)
target_compile_definitions(CMCP PUBLIC C_MCP_ENABLE_VFS=$<BOOL:${CMCP_ENABLE_VFS}>)
target_compile_definitions(CMCP PUBLIC C_MCP_ENABLE_SERIAL_LOOP=$<BOOL:${CMCP_BUILD_SERIAL}>)
target_include_directories(CMCP PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/Config)
if(UNIX)
  target_link_libraries(CMCP PUBLIC m)
endif()
if(CMCP_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()
