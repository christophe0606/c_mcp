cmake_minimum_required(VERSION 3.20)
project(c_mcp LANGUAGES C)

# The core and weak stdio hooks use standard C. HTTP is optional POSIX code.
option(CMCP_BUILD_HTTP "Build the POSIX HTTP transport" ${UNIX})
option(CMCP_BUILD_STDIO "Build the portable stdio input loop" ON)
option(CMCP_BUILD_TESTS "Build portable dispatcher tests" OFF)
option(CMCP_ENABLE_VFS "Enable read-only MCP resources" ON)
add_library(CMCP cJSON.c mcp.c stdio_transport.c)
target_compile_definitions(CMCP PUBLIC C_MCP_ENABLE_VFS=$<BOOL:${CMCP_ENABLE_VFS}>)
target_compile_definitions(CMCP PUBLIC C_MCP_ENABLE_STDIO_LOOP=$<BOOL:${CMCP_BUILD_STDIO}>)
target_include_directories(CMCP PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(UNIX)
  target_link_libraries(CMCP PUBLIC m)
endif()
if(CMCP_BUILD_HTTP)
  target_compile_features(CMCP PRIVATE c_std_11)
endif()
if(CMCP_BUILD_HTTP)
  find_package(Threads REQUIRED)
  target_sources(CMCP PRIVATE http.c)
  target_link_libraries(CMCP PUBLIC Threads::Threads)
endif()
if(CMCP_BUILD_TESTS)
  enable_testing()
  add_executable(cmcp_core_test tests/core_test.c)
  target_link_libraries(cmcp_core_test PRIVATE CMCP)
  add_test(NAME cmcp_core COMMAND cmcp_core_test)
  add_executable(cmcp_core_no_vfs_test tests/core_test.c mcp.c cJSON.c stdio_transport.c)
  target_include_directories(cmcp_core_no_vfs_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_compile_definitions(cmcp_core_no_vfs_test PRIVATE C_MCP_ENABLE_VFS=0)
  target_compile_definitions(cmcp_core_no_vfs_test PRIVATE C_MCP_ENABLE_STDIO_LOOP=0)
  if(UNIX)
    target_link_libraries(cmcp_core_no_vfs_test PRIVATE m)
  endif()
  add_test(NAME cmcp_core_no_vfs COMMAND cmcp_core_no_vfs_test)
  # Separate static library tests real weak/strong symbol resolution, including
  # MSVC alternate-name defaults, independent of the main input-loop option.
  add_library(cmcp_stdio_test_core STATIC mcp.c cJSON.c stdio_transport.c)
  target_include_directories(cmcp_stdio_test_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
  target_compile_definitions(cmcp_stdio_test_core PUBLIC C_MCP_ENABLE_STDIO_LOOP=1)
  target_compile_definitions(cmcp_stdio_test_core PUBLIC C_MCP_ENABLE_VFS=$<BOOL:${CMCP_ENABLE_VFS}>)
  if(UNIX)
    target_link_libraries(cmcp_stdio_test_core PUBLIC m)
  endif()
  add_executable(cmcp_stdio_override_test tests/stdio_override_test.c)
  target_link_libraries(cmcp_stdio_override_test PRIVATE cmcp_stdio_test_core)
  add_test(NAME cmcp_stdio_override COMMAND cmcp_stdio_override_test)
  add_executable(cmcp_stdio_driver tests/stdio_driver.c)
  target_link_libraries(cmcp_stdio_driver PRIVATE cmcp_stdio_test_core)
endif()
